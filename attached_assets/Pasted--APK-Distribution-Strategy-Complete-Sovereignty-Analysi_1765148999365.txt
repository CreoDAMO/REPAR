**Never distribute unsigned APKs.**


## 🚀 AUTOMATED BUILD & DISTRIBUTION PIPELINE

### GitHub Actions Workflow

Add to `.github/workflows/mobile-apk-build.yml`:

```yaml
name: Build & Distribute APK

on:
  push:
    tags:
      - 'mobile-v*'

jobs:
  build-apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20
      
      - name: Install dependencies
        run: cd mobile && npm install
      
      - name: Build APK with EAS
        run: |
          npm install -g eas-cli
          eas build --platform android --profile production --non-interactive
        env:
          EXPO_TOKEN: ${{ secrets.EXPO_TOKEN }}
      
      - name: Download built APK
        run: eas build:download --platform android --latest
      
      - name: Calculate SHA-256
        id: hash
        run: |
          HASH=$(sha256sum aequitas-mobile-*.apk | awk '{print $1}')
          echo "apk_hash=$HASH" >> $GITHUB_OUTPUT
      
      - name: Upload to IPFS
        id: ipfs
        run: |
          # Upload to Pinata or local IPFS node
          IPFS_HASH=$(curl -X POST -F file=@aequitas-mobile-*.apk \
            "https://api.pinata.cloud/pinning/pinFileToIPFS" \
            -H "Authorization: Bearer ${{ secrets.PINATA_JWT }}" \
            | jq -r '.IpfsHash')
          echo "ipfs_hash=$IPFS_HASH" >> $GITHUB_OUTPUT
      
      - name: Anchor to Blockchain
        run: |
          # Submit transaction to x/mobile module
          ./aequitasd tx mobile publish-release \
            --version "${{ github.ref_name }}" \
            --sha256 "${{ steps.hash.outputs.apk_hash }}" \
            --ipfs "${{ steps.ipfs.outputs.ipfs_hash }}" \
            --from founder \
            --chain-id aequitas-1
      
      - name: Create GitHub Release
        uses: softprops/action-gh-release@v1
        with:
          files: aequitas-mobile-*.apk
          body: |
            ## Aequitas Protocol Mobile App ${{ github.ref_name }}
            
            ### Verification
            - **SHA-256:** `${{ steps.hash.outputs.apk_hash }}`
            - **IPFS:** `ipfs://${{ steps.ipfs.outputs.ipfs_hash }}`
            - **Blockchain Record:** Block #TBD
            - **Founder Signature:** ✅ Valid
            
            ### Download Options
            1. **Direct Download:** [GitHub Release](https://github.com/CreoDAMO/REPAR/releases)
            2. **IPFS Gateway:** `https://ipfs.io/ipfs/${{ steps.ipfs.outputs.ipfs_hash }}`
            3. **Website:** https://aequitasprotocol.zone/mobile/download
            
            ### Installation
            1. Download APK
            2. Verify SHA-256 hash matches above
            3. Enable "Install from Unknown Sources" on Android
            4. Install APK
            5. Verify blockchain signature in app
      
      - name: Upload to Website
        run: |
          # SCP to aequitasprotocol.zone
          scp aequitas-mobile-*.apk deploy@aequitasprotocol.zone:/var/www/mobile/
```
