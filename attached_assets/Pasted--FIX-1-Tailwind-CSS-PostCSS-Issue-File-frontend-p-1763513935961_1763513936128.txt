# File: .github/workflows/security-audit.yml
name: Cerberus Security Audit
on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]
  schedule:
    - cron: '0 */6 * * *' # Every 6 hours
jobs:
  security-audit:
    name: Cerberus Security Scan
    runs-on: ubuntu-latest
    timeout-minutes: 45  # Increased from 30
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
        
      - name: Setup Go
        uses: actions/setup-go@v5
        with:
          go-version: '1.22'
          cache: true
          cache-dependency-path: |
            blockchain/go.sum
            auditor/go.sum
            
      - name: Cache Cerberus dependencies
        uses: actions/cache@v4
        with:
          path: |
            ~/go/pkg/mod
            ~/.cache/go-build
            auditor/vendor
          key: ${{ runner.os }}-cerberus-${{ hashFiles('auditor/go.sum') }}
          restore-keys: |
            ${{ runner.os }}-cerberus-
            
      - name: Install Cerberus dependencies
        working-directory: ./auditor
        run: |
          go mod download
          go mod verify
          
      - name: Run Cerberus Orchestrator
        working-directory: ./auditor
        env:
          NVIDIA_API_KEY: ${{ secrets.NVIDIA_API_KEY }}
          CERBERUS_MODE: "full" # full, fast, or critical-only
        run: |
          mkdir -p reports
          go run cmd/cerberus-orchestrator/main.go \
            --target ../blockchain \
            --output reports/cerberus-audit.json \
            --severity critical,high \
            --parallel 4 \
            --timeout 40m
            
      - name: Upload Cerberus Report
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: cerberus-security-report
          path: auditor/reports/*.json
          retention-days: 30
          
      - name: Check for Critical Vulnerabilities
        working-directory: ./auditor
        run: |
          if [ -f reports/cerberus-audit.json ]; then
            CRITICAL=$(jq '.threats | map(select(.severity == "CRITICAL")) | length' reports/cerberus-audit.json)
            if [ "$CRITICAL" -gt 0 ]; then
              echo "❌ CRITICAL vulnerabilities found: $CRITICAL"
              exit 1
            fi
          fi
