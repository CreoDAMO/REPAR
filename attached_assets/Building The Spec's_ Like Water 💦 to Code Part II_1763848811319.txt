    def forward(self, x: np.ndarray, dt: float = 0.01) -> float:  
        """  
        Continuous-time forward pass  
        Differential equation: dx/dt = -x/tau + sigma(Wx + b)

            x: Input vector  
            dt: Time step

### 2.3.2 Liquid Time-Constant Network Specification


```python
# apex/temporal/ltc_spec.py


from dataclasses import dataclass
from typing import List
import numpy as np


@dataclass
class LTCNeuronSpec:
    """Specification for a single Liquid Time-Constant neuron"""
    input_dim: int
    tau: float  # Time constant (0.5 - 2.0)
    num_connections: int = 5  # Sparse connectivity
    weight_init: str = "normal"  # normal, uniform
    weight_scale: float = 0.01


@dataclass
class LTCNetworkSpec:
    """Specification for the complete temporal network"""
    num_neurons: int = 10
    input_dim: int = 50
    tau_range: tuple = (0.5, 2.0)
    output_dim: int = 1
    activation: str = "sigmoid"
    integration_method: str = "euler"  # euler, rk4
    time_step: float = 0.01


class LTCNeuron:
    """
    Liquid Time-Constant Neuron implementation
    Based on: https://arxiv.org/abs/2006.04439
    """
    def __init__(self, spec: LTCNeuronSpec):
        self.spec = spec
        self.tau = spec.tau
        self.state = 0.0
        
        # Sparse random connections
        self.W = np.random.randn(spec.num_connections) * spec.weight_scale
        self.input_indices = np.random.choice(
            spec.input_dim, 
            spec.num_connections, 
            replace=False
        )
    
    def forward(self, x: np.ndarray, dt: float) -> float:
        """
        Differential equation: dx/dt = -x/τ + σ(Wx)
        """
        x_sparse = x[self.input_indices]
        activation = self._sigmoid(np.dot(self.W, x_sparse))
        
        # Continuous-time dynamics
        dx_dt = -self.state / self.tau + activation
        
        # Euler integration
        self.state += dx_dt * dt
        
        return self.state
    
    @staticmethod
    def _sigmoid(x: float) -> float:
        return 1.0 / (1.0 + np.exp(-np.clip(x, -20, 20)))
```

### 2.4.3 Consensus Algorithm


```python
# apex/consensus/algorithm.py


from typing import List, Dict
from dataclasses import dataclass


@dataclass
class ValidatorResult:
    validator_id: str
    verdict: str
    confidence: float
    reputation: float
    reasoning: List[str]


class ByzantineConsensus:
    """
    Byzantine fault-tolerant consensus for APEX verdicts
    """
    
    def __init__(self, required_threshold: float = 0.67):
        self.threshold = required_threshold
    
    def reach_consensus(self, results: List[ValidatorResult]) -> Dict:
        """
        Reach consensus from individual validator results
        
        Algorithm:
        1. Weight each verdict by (confidence * reputation)
        2. Aggregate weighted votes
        3. Check if any verdict exceeds threshold
        4. Return consensus or escalation signal
        """
        
        # Weight by confidence and reputation
        weighted_verdicts = {}
        total_weight = 0.0
        
        for result in results:
            weight = result.confidence * result.reputation
            total_weight += weight
            
            if result.verdict not in weighted_verdicts:
                weighted_verdicts[result.verdict] = {
                    'weight': 0.0,
                    'reasoning': []
                }
            
            weighted_verdicts[result.verdict]['weight'] += weight
            weighted_verdicts[result.verdict]['reasoning'].extend(result.reasoning)
        
        # Find verdict with highest weight
        best_verdict = max(weighted_verdicts, key=lambda v: weighted_verdicts[v]['weight'])
        best_weight = weighted_verdicts[best_verdict]['weight']
        best_confidence = best_weight / total_weight
        
        # Check threshold
        consensus_reached = best_confidence >= self.threshold
        
        return {
            'verdict': best_verdict if consensus_reached else 'escalate',
            'confidence': best_confidence,
            'participating_validators': len(results),
            'reasoning': weighted_verdicts[best_verdict]['reasoning'][:5],  # Top 5
            'requires_escalation': not consensus_reached
        }
```

### 2.5.2 Learning Mechanics


```python
# apex/learning/mechanics.py


from typing import Dict, Any
import numpy as np


class BidirectionalLearning:
    """
    Implements continuous bidirectional learning between humans and APEX
    """
    
    def __init__(self, 
                 constitutional_engine,
                 knowledge_graph,
                 temporal_intelligence):
        self.constitutional = constitutional_engine
        self.knowledge = knowledge_graph
        self.temporal = temporal_intelligence
        
        # Learning rates
        self.alpha_graph = 0.1  # Graph edge update rate
        self.alpha_temporal = 0.05  # Temporal weight update rate
        
    def learn_from_outcome(self, 
                          action: Dict[str, Any],
                          verdict: Dict[str, Any],
                          outcome: Dict[str, Any]):
        """
        Update system based on real-world outcome
        
        This implements the water cycle:
        Human decision (water) → System update (ice forming)
        System insight (ice) → Human understanding (melting)
        """
        
        # 1. Update constitutional precedents
        self.constitutional.learn_from_decision(action, verdict, outcome)
        
        # 2. Update knowledge graph edges
        if outcome['successful']:
            # Strengthen path that led to success
            self._strengthen_reasoning_path(
                verdict['reasoning_path'],
                delta=self.alpha_graph
            )
        else:
            # Weaken and explore alternatives
            self._weaken_reasoning_path(
                verdict['reasoning_path'],
                delta=self.alpha_graph * 0.5
            )
        
        # 3. Update temporal understanding
        # (This happens implicitly through continuous state evolution)
        
    def _strengthen_reasoning_path(self, path: List[str], delta: float):
        """Increase edge weights along successful reasoning path"""
        for i in range(len(path) - 1):
            current_node = path[i]
            next_node = path[i + 1]
            
            current_weight = self.knowledge.G[current_node][next_node]['weight']
            new_weight = min(1.0, current_weight + delta)
            
            self.knowledge.G[current_node][next_node]['weight'] = new_weight
    
    def extract_insights(self, time_range: tuple) -> List[Dict]:
        """
        Extract emergent patterns to share with validators
        
        This is the ice → water phase:
        System discovers patterns in its own operation
        Shares insights with humans
        Humans gain understanding they couldn't reach alone
        """
        insights = []
        
        # Pattern: Frequently co-occurring threats
        threat_patterns = self._find_threat_clusters()
        insights.extend(threat_patterns)
        
        # Pattern: Axioms frequently violated together
        axiom_patterns = self._find_axiom_correlations()
        insights.extend(axiom_patterns)
        
        # Pattern: Temporal trends in threat landscape
        temporal_trends = self._analyze_temporal_evolution()
        insights.extend(temporal_trends)
        
        return insights
```
