get_infrastructure_ip() {

# Function to create or update DNS record
update_dns_record() {
    local name=$1
    local type=$2
    local content=$3
    local proxied=${4:-true}
    local ttl=${5:-1}  # 1 = Auto
    
    echo -e "${BLUE}Updating $name.$DOMAIN → $content${NC}"
    
    # Get existing record ID
    RECORD_ID=$(curl -s -X GET "https://api.cloudflare.com/client/v4/zones/$CLOUDFLARE_ZONE_ID/dns_records?name=$name.$DOMAIN" \
        -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
        -H "Content-Type: application/json" | jq -r '.result[0].id')
    
    if [ "$RECORD_ID" != "null" ] && [ ! -z "$RECORD_ID" ]; then
        # Update existing record
        RESPONSE=$(curl -s -X PUT "https://api.cloudflare.com/client/v4/zones/$CLOUDFLARE_ZONE_ID/dns_records/$RECORD_ID" \
            -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
            -H "Content-Type: application/json" \
            --data "{\"type\":\"$type\",\"name\":\"$name\",\"content\":\"$content\",\"ttl\":$ttl,\"proxied\":$proxied}")
    else
        # Create new record
        RESPONSE=$(curl -s -X POST "https://api.cloudflare.com/client/v4/zones/$CLOUDFLARE_ZONE_ID/dns_records" \
            -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
            -H "Content-Type: application/json" \
            --data "{\"type\":\"$type\",\"name\":\"$name\",\"content\":\"$content\",\"ttl\":$ttl,\"proxied\":$proxied}")
    fi
    
    SUCCESS=$(echo $RESPONSE | jq -r '.success')
    if [ "$SUCCESS" = "true" ]; then
        echo -e "${GREEN}✓ Updated successfully${NC}"
    else
        ERROR=$(echo $RESPONSE | jq -r '.errors[0].message')
        echo -e "${RED}✗ Failed: $ERROR${NC}"
    fi
}

check_system() {
    echo -e "${BLUE}[1/12] Checking system requirements...${NC}"
    
    # Check OS
    if [[ ! -f /etc/os-release ]]; then
        echo -e "${RED}Error: Cannot detect OS${NC}"
        exit 1
    fi
    
    . /etc/os-release
    echo -e "${GREEN}✓ OS: $PRETTY_NAME${NC}"
    
    # Check root/sudo
    if [[ $EUID -ne 0 ]]; then
        echo -e "${YELLOW}⚠ This script requires root/sudo privileges${NC}"
        echo -e "${YELLOW}  Attempting to use sudo...${NC}"
        sudo -v || {
            echo -e "${RED}Error: Cannot obtain sudo privileges${NC}"
            exit 1
        }
        SUDO="sudo"
    else
        SUDO=""
    fi
    
    # Check CPU
    CPU_CORES=$(nproc)
    if [[ $CPU_CORES -lt 4 ]]; then
        echo -e "${YELLOW}⚠ Warning: Only $CPU_CORES CPU cores (recommended: 8+)${NC}"
        read -p "Continue anyway? (y/n) " -n 1 -r

        echo -e "${YELLOW}⚠ ACE API not responding, checking local blockchain...${NC}"
        ACE_AVAILABLE=false
    fi
    
    # Check local blockchain
    if command -v aequitasd &> /dev/null; then
        echo -e "${GREEN}✓ aequitasd: Found${NC}"
        AEQUITASD_AVAILABLE=true
    else
        echo -e "${YELLOW}⚠ aequitasd not in PATH${NC}"
        AEQUITASD_AVAILABLE=false
    fi
    
    # Try to detect RPC endpoint
    RPC_URL=""
    for port in 26657 26658; do
        if curl -s --connect-timeout 2 "http://localhost:$port/status" > /dev/null 2>&1; then
            RPC_URL="http://localhost:$port"
            echo -e "${GREEN}✓ RPC endpoint: $RPC_URL${NC}"
            break
        fi
    done
    
    # Check Cloudflare DNS
    if dig +short rpc.aequitasprotocol.zone > /dev/null 2>&1; then
        PUBLIC_RPC=$(dig +short rpc.aequitasprotocol.zone | head -n 1)
        if [[ ! -z "$PUBLIC_RPC" ]]; then

fetch_chain_data() {

validate_endpoints() {
    echo -e "${BLUE}[7/10] Validating endpoints...${NC}"
    
    local validation_failed=false
    
    # Test RPC
    if [[ ! -z "${ENDPOINTS[rpc]}" ]]; then
        if curl -s --connect-timeout 10 "${ENDPOINTS[rpc]}/status" > /dev/null 2>&1; then
            echo -e "${GREEN}✓ RPC endpoint: Responding${NC}"
        else
            echo -e "${RED}✗ RPC endpoint: Not responding${NC}"
            validation_failed=true
        fi
    fi
    
    # Test REST
    if [[ ! -z "${ENDPOINTS[rest]}" ]]; then
        if curl -s --connect-timeout 10 "${ENDPOINTS[rest]}/cosmos/base/tendermint/v1beta1/node_info" > /dev/null 2>&1; then
            echo -e "${GREEN}✓ REST endpoint: Responding${NC}"
        else
            echo -e "${RED}✗ REST endpoint: Not responding${NC}"
            validation_failed=true
        fi
    fi
    
    # Test Explorer
    if [[ ! -z "${ENDPOINTS[explorer]}" ]]; then
        if curl -s --connect-timeout 10 "${ENDPOINTS[explorer]}" > /dev/null 2>&1; then
            echo -e "${GREEN}✓ Explorer: Accessible${NC}"
        else
            echo -e "${YELLOW}⚠ Explorer: Not accessible${NC}"
        fi
    fi
    
    if [[ $validation_failed == true ]]; then
        echo -e "${YELLOW}⚠ Some endpoints failed validation${NC}"
        echo -e "${YELLOW}  Submission may be rejected if endpoints are not accessible${NC}"
    fi
    
    echo ""
}

                return response.status_code == 200

            self.last_config_hash = new_hash
            return True  # First run
        
        if new_hash != self.last_config_hash:

        self.health_status = self.validate_all_endpoints(self.current_endpoints)

        else:
            self.log("No configuration changes detected", "SUCCESS")
        
        self.log("Check cycle complete", "INFO")
