# home-validator-setup.sh

echo "🏠 Setting up Aequitas Home Validator Node"

# Download binary

# Create systemd service

[Unit]
Description=Aequitas Home Validator

[Service]
Type=simple
User=aequitas-validator
WorkingDirectory=/home/aequitas-validator
ExecStart=/usr/local/bin/aequitasd start
Restart=always
RestartSec=3
LimitNOFILE=4096

[Install]

- name: Deploy Aequitas Home Validators

        name: "{{ item }}"
        state: present
      loop:
        - curl
        - wget
        - jq
  
    - name: Download binary
      get_url:
        url: "{{ binary_url }}"
        dest: "/usr/local/bin/aequitasd"
        mode: '0755'
  
    - name: Create validator user
      user:
        name: aequitas-validator
        shell: /bin/bash
        home: /home/aequitas-validator
  
    - name: Initialize node
      command: "aequitasd init {{ inventory_hostname }} --chain-id {{ chain_id }}"
      become_user: aequitas-validator
  
    - name: Download genesis
      get_url:
        url: "{{ genesis_url }}"
        dest: "/home/aequitas-validator/.aequitas/config/genesis.json"
        owner: aequitas-validator
        group: aequitas-validator
  
    - name: Configure node
      template:
        src: config.toml.j2
        dest: /home/aequitas-validator/.aequitas/config/config.toml
        owner: aequitas-validator
        group: aequitas-validator
  
    - name: Install systemd service
      template:
        src: aequitas-validator.service.j2
        dest: /etc/systemd/system/aequitas-validator.service
  
    - name: Start validator
      systemd:
        name: aequitas-validator
        state: started
        enabled: yes
