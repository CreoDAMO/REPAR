// ace/registry/node_identity.go
type NodeIdentity struct {
    ID          string
    IP          string
    PublicKey   ed25519.PublicKey
    Signature   []byte  // Signed by genesis validator
    GenesisHash string  // Binds to specific chain
}

func (ni *NodeIdentity) Verify() bool {
    // 1. Verify signature matches public key
    if !ed25519.Verify(ni.PublicKey, ni.ID, ni.Signature) {
        return false
    }
    
    // 2. Verify public key is in genesis
    if !genesisHasValidator(ni.PublicKey, ni.GenesisHash) {
        return false
    }
    
    return true
}

// ace/registry/secure_registry.go
type SecureNodeRegistry struct {
    nodes map[string]*NodeIdentity
}

func (sr *SecureNodeRegistry) RegisterNode(node *NodeIdentity) error {
    if !node.Verify() {
        return errors.New("invalid node identity")
    }
    
    sr.nodes[node.ID] = node
    return nil
}
