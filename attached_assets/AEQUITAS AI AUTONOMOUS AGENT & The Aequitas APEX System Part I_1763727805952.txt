package autonomous


import (
    "context"
    "fmt"
    "time"

type AutonomousAgent struct {
    nvidiaClient   *nvidia.NIMClient
    cerberus       *auditor.CerberusOrchestrator
    consensusNode  *consensus.TendermintNode
    threatDB       *ThreatDatabase
    config         *AgentConfig
}


type AgentConfig struct {
    ScanIntervalHours    int
    AutoFixEnabled       bool
    ChaosTestingEnabled  bool
    ThreatThreshold      string // "critical", "high", "medium"
    MaxConcurrentScans   int
    NVIDIAModelID        string
}


func NewAutonomousAgent(cfg *AgentConfig) (*AutonomousAgent, error) {
    nvidiaClient, err := nvidia.NewNIMClient(cfg.NVIDIAModelID)
    if err != nil {
        return nil, fmt.Errorf("nvidia client init failed: %w", err)
    }
    
    cerberus := auditor.NewCerberusOrchestrator()
    
    threatDB, err := NewThreatDatabase()
    if err != nil {
        return nil, fmt.Errorf("threat db init failed: %w", err)
    }
    
    return &AutonomousAgent{
        nvidiaClient:  nvidiaClient,
        cerberus:      cerberus,
        threatDB:      threatDB,
        config:        cfg,
    }, nil
}


func (a *AutonomousAgent) Start(ctx context.Context) error {
    ticker := time.NewTicker(time.Duration(a.config.ScanIntervalHours) * time.Hour)
    defer ticker.Stop()
    
    // Initial scan on startup
    if err := a.runFullCycle(ctx); err != nil {
        return fmt.Errorf("initial scan failed: %w", err)
    }
    
    for {
        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-ticker.C:
            if err := a.runFullCycle(ctx); err != nil {
                // Log error but don't stop agent
                fmt.Printf("cycle error: %v\n", err)
            }
        }
    }
}


func (a *AutonomousAgent) runFullCycle(ctx context.Context) error {
    // Phase 1: Static Analysis (Cerberus)
    threats, err := a.cerberus.ScanBlockchain(ctx)
    if err != nil {
        return fmt.Errorf("cerberus scan failed: %w", err)
    }
    
    // Phase 2: AI Analysis (NVIDIA NIM)
    for _, threat := range threats {
        analysis, err := a.analyzeWithAI(ctx, threat)
        if err != nil {
            fmt.Printf("AI analysis failed for threat %s: %v\n", threat.ID, err)
            continue
        }
        
        threat.AIAnalysis = analysis
        
        // Phase 3: Auto-Fix (if enabled and safe)
        if a.config.AutoFixEnabled && analysis.CanAutoFix {
            if err := a.attemptAutoFix(ctx, threat, analysis); err != nil {
                fmt.Printf("auto-fix failed for threat %s: %v\n", threat.ID, err)
            }
        }
    }
    
    // Phase 4: Chaos Engineering (if enabled)
    if a.config.ChaosTestingEnabled {
        if err := a.runChaosTests(ctx); err != nil {
            fmt.Printf("chaos testing failed: %v\n", err)
        }
    }
    
    // Phase 5: Persist results
    if err := a.threatDB.StoreScanResults(threats); err != nil {
        return fmt.Errorf("failed to persist results: %w", err)
    }
    
    return nil
}


func (a *AutonomousAgent) analyzeWithAI(ctx context.Context, threat *auditor.Threat) (*AIAnalysis, error) {
    prompt := fmt.Sprintf(`Analyze this blockchain security threat:


Threat ID: %s
Severity: %s
Type: %s
Location: %s
Description: %s
Code Snippet:
%s


Provide:
1. Root cause analysis
2. Exploitation difficulty (1-10)
3. Potential impact assessment
4. Can this be auto-fixed safely? (yes/no with reasoning)
5. Recommended fix (detailed steps)
6. Constitutional axiom implications (which of the 25 axioms could be affected?)


Format response as JSON.`, 
        threat.ID, 
        threat.Severity, 
        threat.Type,
        threat.Location,
        threat.Description,
        threat.CodeSnippet,
    )
    
    response, err := a.nvidiaClient.Query(ctx, prompt)
    if err != nil {
        return nil, err
    }
    
    return parseAIAnalysis(response)
}


func (a *AutonomousAgent) attemptAutoFix(ctx context.Context, threat *auditor.Threat, analysis *AIAnalysis) error {
    // Create git branch for fix
    branchName := fmt.Sprintf("autofix/%s", threat.ID)
    
    // Apply fix
    fixPatch, err := a.generateFixPatch(threat, analysis)
    if err != nil {
        return fmt.Errorf("patch generation failed: %w", err)
    }
    
    // Test fix
    testResults, err := a.testFix(ctx, fixPatch)
    if err != nil {
        return fmt.Errorf("fix testing failed: %w", err)
    }
    
    if !testResults.AllTestsPassed {
        return fmt.Errorf("fix tests failed: %v", testResults.Failures)
    }
    
    // Create PR
    prURL, err := a.createFixPR(branchName, threat, analysis, testResults)
    if err != nil {
        return fmt.Errorf("PR creation failed: %w", err)
    }
    
    fmt.Printf("✓ Auto-fix PR created: %s\n", prURL)
    return nil
}


func (a *AutonomousAgent) runChaosTests(ctx context.Context) error {
    chaosScenarios := []ChaosScenario{
        {Name: "Byzantine Node Attack", Severity: "HIGH"},
        {Name: "Network Partition", Severity: "CRITICAL"},
        {Name: "State Corruption", Severity: "CRITICAL"},
        {Name: "DDoS Simulation", Severity: "MEDIUM"},
        {Name: "Consensus Timeout", Severity: "HIGH"},
    }
    
    for _, scenario := range chaosScenarios {
        result, err := a.executeChaosScenario(ctx, scenario)
        if err != nil {
            return fmt.Errorf("chaos scenario %s failed: %w", scenario.Name, err)
        }
        
        if !result.SystemRecovered {
            return fmt.Errorf("CRITICAL: System did not recover from %s", scenario.Name)
        }
        
        // Log chaos test results
        a.threatDB.StoreChaosTestResult(result)
    }
    
    return nil
}

package autonomous


import (
    "database/sql"
    "encoding/json"
    "time"
    
    _ "github.com/lib/pq"
)


type ThreatDatabase struct {
    db *sql.DB
}


func NewThreatDatabase() (*ThreatDatabase, error) {
    db, err := sql.Open("postgres", "postgres://localhost/aequitas_threats?sslmode=disable")
    if err != nil {
        return nil, err
    }
    
    if err := db.Ping(); err != nil {
        return nil, err
    }
    
    // Create tables if not exist
    if err := createTables(db); err != nil {
        return nil, err
    }
    
    return &ThreatDatabase{db: db}, nil
}


func createTables(db *sql.DB) error {
    schema := `
    CREATE TABLE IF NOT EXISTS threats (
        id VARCHAR(255) PRIMARY KEY,
        severity VARCHAR(50) NOT NULL,
        type VARCHAR(100) NOT NULL,
        location TEXT NOT NULL,
        description TEXT NOT NULL,
        code_snippet TEXT,
        detected_at TIMESTAMP NOT NULL DEFAULT NOW(),
        resolved_at TIMESTAMP,
        ai_analysis JSONB,
        auto_fix_attempted BOOLEAN DEFAULT FALSE,
        auto_fix_successful BOOLEAN DEFAULT FALSE,
        pr_url TEXT,
        axioms_affected TEXT[]
    );
    
    CREATE TABLE IF NOT EXISTS chaos_tests (
        id SERIAL PRIMARY KEY,
        scenario_name VARCHAR(255) NOT NULL,
        severity VARCHAR(50) NOT NULL,
        executed_at TIMESTAMP NOT NULL DEFAULT NOW(),
        system_recovered BOOLEAN NOT NULL,
        recovery_time_seconds INT,
        details JSONB
    );
    
    CREATE TABLE IF NOT EXISTS scan_history (
        id SERIAL PRIMARY KEY,
        scan_started_at TIMESTAMP NOT NULL,
        scan_completed_at TIMESTAMP,
        threats_found INT DEFAULT 0,
        threats_fixed INT DEFAULT 0,
        status VARCHAR(50) NOT NULL
    );
    
    CREATE INDEX IF NOT EXISTS idx_threats_severity ON threats(severity);
    CREATE INDEX IF NOT EXISTS idx_threats_detected ON threats(detected_at);
    CREATE INDEX IF NOT EXISTS idx_chaos_executed ON chaos_tests(executed_at);
    `
    
    _, err := db.Exec(schema)
    return err
}


func (db *ThreatDatabase) StoreScanResults(threats []*auditor.Threat) error {
    tx, err := db.db.Begin()
    if err != nil {
        return err
    }
    defer tx.Rollback()
    
    for _, threat := range threats {
        aiAnalysisJSON, _ := json.Marshal(threat.AIAnalysis)
        
        _, err := tx.Exec(`
            INSERT INTO threats (
                id, severity, type, location, description, 
                code_snippet, ai_analysis, axioms_affected
            ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
            ON CONFLICT (id) DO UPDATE SET
                ai_analysis = EXCLUDED.ai_analysis,
                resolved_at = CASE 
                    WHEN threats.resolved_at IS NULL AND EXCLUDED.auto_fix_successful = true 
                    THEN NOW() 
                    ELSE threats.resolved_at 
                END
        `, threat.ID, threat.Severity, threat.Type, threat.Location,
           threat.Description, threat.CodeSnippet, aiAnalysisJSON,
           threat.AxiomsAffected)
           
        if err != nil {
            return err
        }
    }
    
    return tx.Commit()
}

package main


import (
    "context"
    "flag"
    "fmt"
    "os"
    "os/signal"
    "syscall"
    
    "github.com/REPAR/aequitas/ai/autonomous"
)


func main() {
    var (
        scanInterval    = flag.Int("interval", 6, "Scan interval in hours")
        autoFix         = flag.Bool("autofix", true, "Enable auto-fix")
        chaosEnabled    = flag.Bool("chaos", true, "Enable chaos engineering")
        threatThreshold = flag.String("threshold", "high", "Minimum threat severity")
        nvidiaModel     = flag.String("model", "aequitas-unified-1.0", "NVIDIA NIM model ID")
    )
    flag.Parse()
    
    config := &autonomous.AgentConfig{
        ScanIntervalHours:    *scanInterval,
        AutoFixEnabled:       *autoFix,
        ChaosTestingEnabled:  *chaosEnabled,
        ThreatThreshold:      *threatThreshold,
        MaxConcurrentScans:   4,
        NVIDIAModelID:        *nvidiaModel,
    }
    
    agent, err := autonomous.NewAutonomousAgent(config)
    if err != nil {
        fmt.Fprintf(os.Stderr, "Failed to create agent: %v\n", err)
        os.Exit(1)
    }
    
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    
    // Handle shutdown gracefully
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, os.Interrupt, syscall.SIGTERM)
    
    go func() {
        <-sigChan
        fmt.Println("\nShutting down autonomous agent...")
        cancel()
    }()
    
    fmt.Println("🤖 Aequitas Autonomous AI Agent starting...")
    fmt.Printf("   Scan interval: %d hours\n", *scanInterval)
    fmt.Printf("   Auto-fix: %v\n", *autoFix)
    fmt.Printf("   Chaos testing: %v\n", *chaosEnabled)
    
    if err := agent.Start(ctx); err != nil && err != context.Canceled {
        fmt.Fprintf(os.Stderr, "Agent error: %v\n", err)
        os.Exit(1)
    }
    
    fmt.Println("Agent shutdown complete.")
}

### Step 3: Integrate with NVIDIA NIM (15 minutes)


```bash
# Create NVIDIA client if doesn't exist
mkdir -p ai/nvidia


cat > ai/nvidia/client.go << 'EOF'
package nvidia


import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "net/http"
    "os"
)


type NIMClient struct {
    apiKey   string
    modelID  string
    endpoint string
}


func NewNIMClient(modelID string) (*NIMClient, error) {
    apiKey := os.Getenv("NVIDIA_API_KEY")
    if apiKey == "" {
        return nil, fmt.Errorf("NVIDIA_API_KEY environment variable not set")
    }
    
    return &NIMClient{
        apiKey:   apiKey,
        modelID:  modelID,
        endpoint: "https://api.nvidia.com/v1/nim/inference",
    }, nil
}


type QueryRequest struct {
    Model  string `json:"model"`
    Prompt string `json:"prompt"`
}


type QueryResponse struct {
    Choices []struct {
        Text string `json:"text"`
    } `json:"choices"`
}


func (c *NIMClient) Query(ctx context.Context, prompt string) (string, error) {
    reqBody := QueryRequest{
        Model:  c.modelID,
        Prompt: prompt,
    }
    
    jsonData, err := json.Marshal(reqBody)
    if err != nil {
        return "", err
    }
    
    req, err := http.NewRequestWithContext(ctx, "POST", c.endpoint, bytes.NewBuffer(jsonData))
    if err != nil {
        return "", err
    }
    
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", fmt.Sprintf("Bearer %s", c.apiKey))
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return "", fmt.Errorf("NVIDIA API returned status %d", resp.StatusCode)
    }
    
    var response QueryResponse
    if err := json.NewDecoder(resp.Body).Decode(&response); err != nil {
        return "", err
    }
    
    if len(response.Choices) == 0 {
        return "", fmt.Errorf("no response from NVIDIA API")
    }
    
    return response.Choices[0].Text, nil
}
EOF
```

class AdversarialDetector:
    """
    DARPA GARD-inspired adversarial attack detector
    Detects noise patterns that might cause AI misclassification
    """
    
    def __init__(self):
        self.baseline_entropy = None
        self.detection_threshold = 0.3
    
    def detect(self, image: np.ndarray) -> bool:
        """
        Detect adversarial perturbations in sensor data
        Returns: True if adversarial attack detected
        """
        # Calculate image entropy
        entropy = self.calculate_entropy(image)
        
        if self.baseline_entropy is None:
            self.baseline_entropy = entropy
            return False
        
        # Compare to baseline
        entropy_delta = abs(entropy - self.baseline_entropy)
        
        if entropy_delta > self.detection_threshold:
            return True  # Adversarial attack likely
        
        return False
    
    def calculate_entropy(self, image: np.ndarray) -> float:
        """Calculate Shannon entropy of image"""
        histogram = np.histogram(image, bins=256)[0]
        histogram = histogram / histogram.sum()
        entropy = -np.sum(histogram * np.log2(histogram + 1e-10))
        return entropy

@dataclass

    """Swarm agent metadata"""
    id: str
    role: SwarmRole
    capabilities: AgentCapabilities
    pose: Optional[Tuple[float, float]] = None

    task_history: List[Dict] = field(default_factory=list)




class SwarmOrchestrator(Node):
    """
    OpenAI Swarm-inspired multi-agent orchestrator
    + Consensus-Based Optimization
    + Stigmergic Communication (pheromone-like)
    """
    
    def __init__(self, num_agents: int = 10):
        super().__init__('swarm_orchestrator')
        
        self.num_agents = num_agents
        self.agents: Dict[str, SwarmAgent] = {}
        self.pheromone_map: Dict[Tuple[int, int], float] = {}  # Stigmergic communication
        self.constitutional_enforcer = ConstitutionalEnforcer()
        
        # Swarm communication
        self.command_pub = self.create_publisher(String, '/swarm/commands', 10)
        self.state_sub = self.create_subscription(
            String, '/swarm/agent_state', self.agent_state_callback, 10
        )
        
        # Mission state
        self.active_missions: List[Dict] = []
        self.completed_missions: List[Dict] = []
        
        # Orchestration timer
        self.orchestration_timer = self.create_timer(1.0, self.orchestrate)
        
        print(f"✅ Swarm Orchestrator initialized for {num_agents} agents")
    
    def agent_state_callback(self, msg: String):
        """Receive agent state updates"""
        try:
            state = json.loads(msg.data)
            agent_id = state['agent_id']
            
            if agent_id not in self.agents:
                # Register new agent
                self.agents[agent_id] = SwarmAgent(
                    id=agent_id,
                    role=SwarmRole.SCOUT,  # Default role
                    capabilities=AgentCapabilities()
                )
            
            # Update agent state
            self.agents[agent_id].pose = (
                state['pose']['x'],
                state['pose']['y']
            )
            self.agents[agent_id].status = state['status']
            
        except (json.JSONDecodeError, KeyError) as e:
            self.get_logger().error(f"Invalid agent state: {e}")
    
    def orchestrate(self):
        """
        Main swarm orchestration loop
        Implements: Task allocation, consensus, stigmergic coordination
        """
        if len(self.agents) == 0:
            return
        
        # 1. Task allocation (using swarm intelligence principles)
        idle_agents = [a for a in self.agents.values() if a.status == "IDLE"]
        
        if idle_agents and self.active_missions:
            mission = self.active_missions[0]
            
            # Select best agent using consensus-based optimization
            best_agent = self.select_agent_consensus(idle_agents, mission)
            
            # Assign task
            self.assign_task(best_agent, mission)
        
        # 2. Update pheromone map (stigmergic communication)
        self.update_pheromones()
        
        # 3. Check for emergent behaviors
        self.detect_emergent_behaviors()
    
    def select_agent_consensus(self, agents: List[SwarmAgent], mission: Dict) -> SwarmAgent:
        """
        Consensus-Based Optimization for agent selection
        Each agent has a "fitness" for the mission, swarm converges to consensus
        """
        # Calculate fitness scores
        fitness_scores = {}
        for agent in agents:
            score = self.calculate_agent_fitness(agent, mission)
            fitness_scores[agent.id] = score
        
        # Consensus: weighted average with neighbor influence
        consensus_scores = {}
        for agent in agents:
            neighbor_scores = [
                fitness_scores[a.id] for a in agents 
                if self.are_neighbors(agent, a)
            ]
            
            if neighbor_scores:
                consensus_scores[agent.id] = (
                    0.7 * fitness_scores[agent.id] +
                    0.3 * np.mean(neighbor_scores)
                )
            else:
                consensus_scores[agent.id] = fitness_scores[agent.id]
        
        # Select agent with highest consensus score
        best_agent_id = max(consensus_scores, key=consensus_scores.get)
        return next(a for a in agents if a.id == best_agent_id)
    
    def calculate_agent_fitness(self, agent: SwarmAgent, mission: Dict) -> float:
        """Calculate agent fitness for mission"""
        score = 0.0
        
        # Distance to mission location
        if agent.pose and 'location' in mission:
            dist = distance.euclidean(agent.pose, mission['location'])
            score += 1.0 / (1.0 + dist)  # Closer = higher score
        
        # Capability match
        required_caps = mission.get('required_capabilities', [])
        agent_caps = agent.capabilities.perception + agent.capabilities.navigation
        matches = sum(1 for cap in required_caps if cap in agent_caps)
        score += matches * 0.5
        
        # Task history (experience bonus)
        relevant_tasks = sum(
            1 for task in agent.task_history 
            if task.get('type') == mission.get('type')
        )
        score += relevant_tasks * 0.2
        
        return score
    
    def are_neighbors(self, agent1: SwarmAgent, agent2: SwarmAgent) -> bool:
        """Check if agents are neighbors (within communication range)"""
        if not (agent1.pose and agent2.pose):
            return False
        
        dist = distance.euclidean(agent1.pose, agent2.pose)
        return dist < 10.0  # 10 meter communication range
    
    def assign_task(self, agent: SwarmAgent, mission: Dict):
        """Assign task to agent"""
        # Constitutional validation
        task_action = {
            'type': mission['type'],
            'target': mission.get('target'),
            'percentage': mission.get('percentage', 100)
        }
        
        if not self.constitutional_enforcer.validate_action(task_action, agent.id):
            self.get_logger().error(f"❌ Task rejected: Constitutional violation")
            return
        
        # Send command to agent
        command = {
            'type': mission['type'],
            'agent_id': agent.id,
            **mission
        }
        
        self.command_pub.publish(String(data=json.dumps(command)))
        
        # Update agent state
        agent.status = "ASSIGNED"
        agent.task_history.append(mission)
        
        self.get_logger().info(f"✅ Task assigned to agent {agent.id}: {mission['type']}")
    
    def update_pheromones(self):
        """
        Stigmergic communication: Update pheromone map
        Agents leave "pheromones" (virtual markers) that guide other agents
        """
        # Evaporate pheromones over time
        for cell in list(self.pheromone_map.keys()):
            self.pheromone_map[cell] *= 0.95  # 5% evaporation
            if self.pheromone_map[cell] < 0.01:
                del self.pheromone_map[cell]
        
        # Deposit new pheromones at agent locations
        for agent in self.agents.values():
            if agent.pose:
                cell = (int(agent.pose[0]), int(agent.pose[1]))
                self.pheromone_map[cell] = self.pheromone_map.get(cell, 0) + 1.0
    
    def detect_emergent_behaviors(self):
        """
        Detect emergent swarm behaviors
        Examples: Formation, clustering, coordinated search patterns
        """
        if len(self.agents) < 3:
            return
        
        # Detect formation patterns
        poses = [a.pose for a in self.agents.values() if a.pose]
        if len(poses) >= 3:
            # Check if agents are in formation (e.g., line, circle)
            poses_array = np.array(poses)
            
            # DBSCAN clustering
            clustering = DBSCAN(eps=5.0, min_samples=2).fit(poses_array)
            num_clusters = len(set(clustering.labels_)) - (1 if -1 in clustering.labels_ else 0)
            
            if num_clusters == 1:
                self.get_logger().info(f"🔄 Emergent behavior: Swarm clustering detected")

class SwarmRLPolicy(nn.Module):
    """
    Multi-Agent Reinforcement Learning Policy Network
    Trained to optimize: enforcement efficiency, constitutional compliance, system resilience
    """
    
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 256):
        super().__init__()
        
        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh()
        )
        
        self.value_head = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, state: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Forward pass
        Returns: (action_logits, state_value)
        """
        action_logits = self.network(state)
        value = self.value_head(state)
        return action_logits, value




class SwarmRLTrainer:
    """
    Trains swarm policy using Multi-Agent Proximal Policy Optimization (MAPPO)
    Reward function optimized for:
    - Reparations enforcement speed
    - Constitutional axiom compliance
    - Byzantine fault tolerance
    - Energy efficiency
    """
    
    def __init__(self, num_agents: int, state_dim: int, action_dim: int):
        self.num_agents = num_agents
        self.policy = SwarmRLPolicy(state_dim, action_dim)
        self.optimizer = torch.optim.Adam(self.policy.parameters(), lr=3e-4)
        
        self.replay_buffer = deque(maxlen=10000)
        self.training_step = 0
        
    def calculate_reward(self, agent_state: Dict, action: Dict, next_state: Dict) -> float:
        """
        Calculate reward for agent action
        Incorporates constitutional compliance and mission objectives
        """
        reward = 0.0
        
        # Reward for successful enforcement
        if action.get('type') == 'enforce_reparations' and next_state.get('enforcement_successful'):
            reward += 100.0
        
        # Penalty for constitutional violations
        if action.get('constitutional_violation'):
            reward -= 500.0  # MASSIVE penalty
        
        # Reward for speed (time to complete mission)
        mission_time = next_state.get('mission_duration', 0)
        if mission_time > 0:
            reward += 50.0 / mission_time  # Faster = higher reward
        
        # Penalty for energy consumption (encourage efficiency)
        energy_used = next_state.get('energy_consumed', 0)
        reward -= 0.1 * energy_used
        
        # Reward for team coordination
        if next_state.get('coordinated_with_neighbors'):
            reward += 20.0
        
        # Penalty for collisions
        if next_state.get('collision'):
            reward -= 50.0
        
        return reward
    
    def train_step(self, batch_size: int = 128) -> float:
        """
        Perform one training step using MAPPO
        Returns: average loss
        """
        if len(self.replay_buffer) < batch_size:
            return 0.0
        
        # Sample batch
        batch = np.random.choice(self.replay_buffer, batch_size, replace=False)
        
        states = torch.tensor([b['state'] for b in batch], dtype=torch.float32)
        actions = torch.tensor([b['action'] for b in batch], dtype=torch.float32)
        rewards = torch.tensor([b['reward'] for b in batch], dtype=torch.float32)
        next_states = torch.tensor([b['next_state'] for b in batch], dtype=torch.float32)
        dones = torch.tensor([b['done'] for b in batch], dtype=torch.float32)
        
        # Forward pass
        action_logits, values = self.policy(states)
        _, next_values = self.policy(next_states)
        
        # Calculate advantages (Generalized Advantage Estimation)
        advantages = rewards + 0.99 * next_values.squeeze() * (1 - dones) - values.squeeze()
        
        # Policy loss (PPO clip)
        action_probs = torch.softmax(action_logits, dim=-1)
        old_action_probs = action_probs.detach()
        
        ratio = action_probs / (old_action_probs + 1e-8)
        clipped_ratio = torch.clamp(ratio, 0.8, 1.2)
        
        policy_loss = -torch.min(
            ratio * advantages,
            clipped_ratio * advantages
        ).mean()
        
        # Value loss
        value_loss = nn.MSELoss()(values.squeeze(), rewards + 0.99 * next_values.squeeze() * (1 - dones))
        
        # Total loss
        loss = policy_loss + 0.5 * value_loss
        
        # Backprop
        self.optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.policy.parameters(), 0.5)
        self.optimizer.step()
        
        self.training_step += 1
        
        return loss.item()
    
    def store_transition(self, state: np.ndarray, action: np.ndarray, 
                         reward: float, next_state: np.ndarray, done: bool):
        """Store transition in replay buffer"""
        self.replay_buffer.append({
            'state': state,
            'action': action,
            'reward': reward,
            'next_state': next_state,
            'done': done
        })
