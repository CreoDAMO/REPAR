1. **Cosmos SDK Blockchain Client (Wiring to Identity/Storage)**:
   - Extend blockchain/client.go (add protobuf, gas, sequencing).
   ```go
   package blockchain

   import (
       "context"
       "fmt"
       "github.com/cosmos/cosmos-sdk/client"
       "github.com/cosmos/cosmos-sdk/client/tx"
       "github.com/cosmos/cosmos-sdk/types"
       "github.com/cosmos/cosmos-sdk/types/tx/signing"
       "github.com/cosmos/ledger-go"
       "github.com/cosmos/cosmos-sdk/x/auth/signing"
   )

   type CosmosClient struct {
       Client client.Context
   }

   func NewCosmosClient(rpcURL string) *CosmosClient {
       cliCtx := client.Context{}.WithNodeURI(rpcURL).WithInterfaceRegistry(types.NewInterfaceRegistry())
       return &CosmosClient{Client: cliCtx}
   }

   func (c *CosmosClient) SendTx(userDID string, msg types.Msg, privKey ledger.PrivKey) (string, error) {
       acc, err := c.Client.GetAccount(context.Background(), userDID)
       if err != nil { return "", err }
       txBuilder := c.Client.TxConfig.NewTxBuilder()
       txBuilder.SetMsgs(msg)
       txBuilder.SetGasLimit(200000) // Adjust based on tx
       txBuilder.SetFeeAmount(types.NewCoins(types.NewInt64Coin("repar", 1000)))
       sigsV2 := signing.SignatureV2{ PubKey: privKey.PubKey(), Data: &signing.SingleSignatureData{ SignMode: signing.SignMode_SIGN_MODE_DIRECT, Signature: nil } }
       txBuilder.SetSignatures(sigsV2)
       signerData := authsigning.SignerData{ ChainID: "aequitas-1", AccountNumber: acc.GetAccountNumber(), Sequence: acc.GetSequence() }
       sig, err := tx.SignWithPrivKey(context.Background(), signing.SignMode_SIGN_MODE_DIRECT, signerData, txBuilder, privKey, c.Client.TxConfig.SignModeHandler(), acc.GetSequence())
       if err != nil { return "", err }
       txBuilder.SetSignatures(sig)
       txBytes, err := c.Client.TxConfig.TxEncoder()(txBuilder.GetTx())
       if err != nil { return "", err }
       res, err := c.Client.BroadcastTxSync(context.Background(), txBytes)
       if err != nil { return "", err }
       return res.TxHash, nil
   }

   func (c *CosmosClient) RecordAllocation(userDID string, optimalNode string, request ResourceRequest) string {
       msg := types.NewMsgRecordAllocation(userDID, optimalNode, request.Type) // Custom msg from x/justice
       txHash, _ := c.SendTx(userDID, msg, GetPrivKeyFromDID(userDID)) // From identity engine
       return txHash
   }
   ```
