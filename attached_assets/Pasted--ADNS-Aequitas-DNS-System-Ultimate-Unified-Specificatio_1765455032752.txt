    // Layer 1: Check Redis cache first (99% hit rate)
    cacheKey := fmt.Sprintf("adns:%s", domain)
    cached, err := k.redisClient.Get(context.Background(), cacheKey).Bytes()
    if err == nil {
        var record types.DNSRecord
        k.cdc.MustUnmarshal(cached, &record)
        return &record, nil
    }
    
    // Layer 2: Cache miss - query blockchain
    store := ctx.KVStore(k.storeKey)
    key := types.DNSRecordKey(domain)
    value := store.Get(key)
    if value == nil {
        return nil, fmt.Errorf("domain not found: %s", domain)

    k.cdc.MustUnmarshal(value, &record)
    
    // Cache for TTL duration
    k.redisClient.Set(
        context.Background(),
        cacheKey,
        value,

    return &record, nil

package main

import (
    "context"
    "flag"
    "fmt"
    "log"
    "os"
    "os/signal"
    "syscall"
    
    "github.com/miekg/dns"
    "github.com/redis/go-redis/v9"
    
    "github.com/CreoDAMO/REPAR/adns-server/cache"
    "github.com/CreoDAMO/REPAR/adns-server/authority"
    "github.com/CreoDAMO/REPAR/adns-server/fallback"
)

type ADNSServer struct {
    cache     *cache.RedisCache
    authority *authority.BlockchainAuthority
    fallback  *fallback.MultiLayerFallback
}

func main() {
    var (
        port      = flag.Int("port", 53, "DNS port")
        rpcNode   = flag.String("rpc", "http://localhost:26657", "Aequitas RPC")
        redisAddr = flag.String("redis", "localhost:6379", "Redis address")
    )
    flag.Parse()
    
    // Initialize components
    rdb := redis.NewClient(&redis.Options{Addr: *redisAddr})
    cacheLayer := cache.NewRedisCache(rdb)
    auth := authority.NewBlockchainAuthority(*rpcNode)
    fb := fallback.NewMultiLayerFallback()
    
    server := &ADNSServer{
        cache:     cacheLayer,
        authority: auth,
        fallback:  fb,
    }
    
    // Start DNS server
    dns.HandleFunc(".", server.handleDNS)
    
    udpServer := &dns.Server{Addr: fmt.Sprintf(":%d", *port), Net: "udp"}
    tcpServer := &dns.Server{Addr: fmt.Sprintf(":%d", *port), Net: "tcp"}
    
    go func() {
        log.Printf("🌍 Starting UDP DNS server on port %d\n", *port)
        if err := udpServer.ListenAndServe(); err != nil {
            log.Fatalf("Failed to start UDP server: %v\n", err)
        }
    }()
    
    go func() {
        log.Printf("🌍 Starting TCP DNS server on port %d\n", *port)
        if err := tcpServer.ListenAndServe(); err != nil {
            log.Fatalf("Failed to start TCP server: %v\n", err)
        }
    }()
    
    // Wait for interrupt
    sig := make(chan os.Signal, 1)
    signal.Notify(sig, syscall.SIGINT, syscall.SIGTERM)
    <-sig
    
    log.Println("Shutting down...")
    udpServer.Shutdown()
    tcpServer.Shutdown()
}

func (s *ADNSServer) handleDNS(w dns.ResponseWriter, r *dns.Msg) {
    m := new(dns.Msg)
    m.SetReply(r)
    m.Authoritative = true
    
    for _, q := range r.Question {
        log.Printf("Query: %s %s\n", q.Name, dns.TypeToString[q.Qtype])
        
        // Layer 1: Check cache (99% hit rate, <1ms)
        if answer := s.cache.Get(q.Name, q.Qtype); answer != nil {
            m.Answer = append(m.Answer, answer)
            log.Printf("✅ Cache hit: %s\n", q.Name)
            w.WriteMsg(m)
            return
        }
        
        // Layer 2: Query blockchain
        if answer, err := s.authority.Resolve(q.Name, q.Qtype); err == nil {
            m.Answer = append(m.Answer, answer)
            s.cache.Set(q.Name, q.Qtype, answer)
            log.Printf("⛓️  Blockchain resolve: %s\n", q.Name)
            w.WriteMsg(m)
            return
        }
        
        // Layer 3: Multi-layer fallback
        if answer := s.fallback.Resolve(q.Name, q.Qtype); answer != nil {
            m.Answer = append(m.Answer, answer)
            log.Printf("🔄 Fallback resolve: %s\n", q.Name)
            w.WriteMsg(m)
            return
        }
        
        // Not found
        m.Rcode = dns.RcodeNameError
    }
    
    w.WriteMsg(m)
}

func (c *RedisCache) Get(domain string, qtype uint16) dns.RR {
    key := fmt.Sprintf("adns:%s:%d", domain, qtype)
    val, err := c.client.Get(context.Background(), key).Result()
    if err != nil {
        return nil
    }
    
    rr, _ := dns.NewRR(val)
    return rr
}

    key := fmt.Sprintf("adns:%s:%d", domain, qtype)
    ttl := time.Duration(answer.Header().Ttl) * time.Second
    c.client.Set(context.Background(), key, answer.String(), ttl)
}

package fallback

import (
    "context"
    "fmt"
    "time"
    
    "github.com/miekg/dns"
)

type MultiLayerFallback struct {
    layers []Resolver
}

type Resolver interface {
    Resolve(domain string, qtype uint16) (dns.RR, error)
    Timeout() time.Duration
    Name() string
}

func NewMultiLayerFallback() *MultiLayerFallback {
    return &MultiLayerFallback{
        layers: []Resolver{
            &TraditionalDNS{},
            &HandshakeResolver{},
            &ENSResolver{},
            &IPFSResolver{},
            &NostrResolver{},
            &TorResolver{},
            &LibP2PResolver{},
            &LoRaResolver{},
        },
    }
}

func (m *MultiLayerFallback) Resolve(domain string, qtype uint16) dns.RR {
    for _, layer := range m.layers {
        ctx, cancel := context.WithTimeout(context.Background(), layer.Timeout())
        defer cancel()
        
        done := make(chan dns.RR, 1)
        go func() {
            if rr, err := layer.Resolve(domain, qtype); err == nil {
                done <- rr
            } else {
                done <- nil
            }
        }()
        
        select {
        case rr := <-done:
            if rr != nil {
                fmt.Printf("✅ %s resolved via %s\n", domain, layer.Name())
                return rr
            }
        case <-ctx.Done():
            fmt.Printf("⏱️  %s timeout on %s\n", layer.Name(), domain)
            continue
        }
    }
    return nil
}

import AsyncStorage from '@react-native-async-storage/async-storage';

export class MobileCache {
  async get(key: string): Promise<string | null> {
    try {
      const value = await AsyncStorage.getItem(`adns:${key}`);
      if (value) {
        const parsed = JSON.parse(value);
        if (Date.now() < parsed.expires) {
          return parsed.data;
        }
        // Expired - remove
        await AsyncStorage.removeItem(`adns:${key}`);
      }
    } catch (e) {
      console.error('Cache get error:', e);
    }
    return null;
  }

  async set(key: string, value: string, ttl: number): Promise<void> {
    try {
      await AsyncStorage.setItem(
        `adns:${key}`,
        JSON.stringify({
          data: value,
          expires: Date.now() + ttl * 1000,
        })
      );
    } catch (e) {
      console.error('Cache set error:', e);
    }
  }
}
