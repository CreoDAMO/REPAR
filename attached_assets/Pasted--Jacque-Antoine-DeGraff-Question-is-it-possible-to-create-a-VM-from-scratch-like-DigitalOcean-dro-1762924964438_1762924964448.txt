🛠️ Understanding the Software Ecosystem

These tools often work together in a layered ecosystem:

· The Hypervisor: This is the foundational layer that allows multiple operating systems to run on a single physical machine. KVM and the Xen Project are examples of open-source hypervisors.
· Management Tools: For a more complete experience, platforms like Proxmox VE and oVirt bundle the hypervisor with a user-friendly management interface, handling tasks like storage, networking, and high availability.

💡 How to Get Started

Your choice should depend on your goals and technical comfort level:

· For Beginners and Home Labs: Start with Proxmox VE. Its integrated nature and web interface make it straightforward to install and manage.
· For Learning Core Virtualization: Experiment with the KVM/QEMU/Libvirt stack on a Linux system. Using a GUI like virt-manager can make this process more intuitive.

// frontend/src/services/blockchain.ts
import { SigningStargateClient, StargateClient } from '@cosmjs/stargate';
import { DirectSecp256k1HdWallet } from '@cosmjs/proto-signing';

export class AequitasClient {
  private client: StargateClient | null = null;
  
  async connect(rpcEndpoint: string) {
    this.client = await StargateClient.connect(rpcEndpoint);
  }
  
  async getBalance(address: string): Promise<string> {
    const balance = await this.client!.getBalance(address, 'urepar');
    return (parseInt(balance.amount) / 1_000_000).toFixed(6);
  }
  
  async sendTokens(
    mnemonic: string,
    recipient: string,
    amount: string,
    memo: string = ''
  ): Promise<string> {
    const wallet = await DirectSecp256k1HdWallet.fromMnemonic(mnemonic, { prefix: 'repar' });
    const [{ address: sender }] = await wallet.getAccounts();
    
    const signingClient = await SigningStargateClient.connectWithSigner(
      this.rpcEndpoint,
      wallet
    );
    
    const result = await signingClient.sendTokens(
      sender,
      recipient,
      [{ denom: 'urepar', amount: (parseFloat(amount) * 1_000_000).toString() }],
      'auto',
      memo
    );
    
    return result.transactionHash;
  }
  
  async voteOnProposal(
    mnemonic: string,
    proposalId: number,
    option: 'yes' | 'no' | 'abstain' | 'no_with_veto'
  ): Promise<string> {
    const wallet = await DirectSecp256k1HdWallet.fromMnemonic(mnemonic, { prefix: 'repar' });
    const [{ address: voter }] = await wallet.getAccounts();
    
    const signingClient = await SigningStargateClient.connectWithSigner(
      this.rpcEndpoint,
      wallet
    );
    
    const msg = {
      typeUrl: '/cosmos.gov.v1beta1.MsgVote',
      value: {
        proposalId: proposalId,
        voter: voter,
        option: this.mapVoteOption(option)
      }
    };
    
    const result = await signingClient.signAndBroadcast(voter, [msg], 'auto');
    return result.transactionHash;
  }
