#!/bin/bash


# Query chain for records and generate zone files
aequitasd query adns list-domains --output json > records.json


# Parse JSON and generate BIND zone files (use jq or similar)
jq '.records[] | select(.domain | endswith(".aequitas"))' records.json | while read record; do
    domain=$(echo $record | jq '.domain')
    type=$(echo $record | jq '.record_type')
    value=$(echo $record | jq '.values[0]')
    echo "$domain 300 IN $type $value" >> zones/db.aequitas
done


# Sign zones
rndc sign aequitas
rndc sign repar
rndc sign sovereign
