Mobile Node Setup Script:

```bash
#!/bin/bash
# aequitas-mobile-node.sh

echo "📱 Starting Aequitas Mobile Sovereign Node"

# Auto-detect platform
if [[ "$OSTYPE" == "linux-android" ]]; then
    BINARY="aequitasd-android-arm64"
elif [[ "$OSTYPE" == "darwin"* ]]; then
    BINARY="aequitasd-ios-arm64" 
else
    BINARY="aequitasd-linux-arm64"
fi

# Download mobile-optimized binary
wget "https://github.com/CreoDAMO/REPAR/releases/download/v1.0.0/${BINARY}" -O aequitasd
chmod +x aequitasd

# Initialize with mobile profile
./aequitasd init "mobile-$(date +%s)" --chain-id aequitas-1 --mode=light

# Download mobile-optimized genesis
wget -O ~/.aequitas/config/genesis.json \
    "https://raw.githubusercontent.com/CreoDAMO/REPAR/main/bin/genesis-mobile.json"

# Configure for mobile
cat > ~/.aequitas/config/mobile-config.toml << EOF
[mobile]
  enabled = true
  battery_saver = true
  data_saver = true
  sync_interval = "5m"
  
[p2p]
  max_connections = 25
  persistent_peers = "core1.aequitas.zone:26656,core2.aequitas.zone:26656"
  
[rpc]
  laddr = "tcp://127.0.0.1:26657"
  
[statesync]
  enable = true
  rpc_servers = "https://rpc.aequitas.zone:26657,https://rpc2.aequitas.zone:26657"
EOF

echo "✅ Mobile node ready - starting background service"
./aequitasd start --home ~/.aequitas --mode=light
```
