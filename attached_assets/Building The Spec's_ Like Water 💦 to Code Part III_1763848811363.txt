class LiquidNeuron:
    """
    Liquid Time-Constant Neuron (MIT)
    Continuous-time dynamics
    """
    def __init__(self, input_dim: int, tau: float = 1.0):
        self.tau = tau
        self.state = 0.0
        self.W = np.random.randn(5) * 0.01
        self.input_indices = np.random.choice(input_dim, 5, replace=False)
    
    def evolve(self, x: np.ndarray, dt: float = 0.01) -> float:
        """
        Differential equation: dx/dt = -x/τ + σ(Wx)
        """
        x_sparse = x[self.input_indices]
        activation = 1.0 / (1.0 + np.exp(-np.dot(self.W, x_sparse)))
        
        dx_dt = -self.state / self.tau + activation
        self.state += dx_dt * dt
        
        return self.state
