        this.device.onReceive((packet) => {
            this.handleValidatorMessage(packet);
        });

        const compressed = this.compress(header);

        );

        if (packet.channel === 'AEQUITAS_VALIDATOR') {
            const header = this.decompress(packet.payload);
            this.processBlockHeader(header);
        }
    }
}

    BluetoothPeers   []string

func (m *MeshValidator) BroadcastBlock(block types.Block) error {
    // Try all available mesh protocols
    errors := []error{}
    
    if m.LoRaEnabled {
        if err := m.broadcastViaLoRa(block); err != nil {
            errors = append(errors, err)
        }
    }
    
    if len(m.WiFiDirectPeers) > 0 {
        if err := m.broadcastViaWiFi(block); err != nil {
            errors = append(errors, err)
        }
    }
    
    // Return success if ANY broadcast succeeded
    if len(errors) < 3 {  // At least one succeeded
        return nil
    }
    return fmt.Errorf("all mesh broadcasts failed")
}

// aequitas/satellite/protocol.go
package satellite

import "github.com/pkg/ax25"

type SatelliteUplink struct {
    Callsign string  // Amateur radio callsign
    Frequency float64
    Station *ax25.Session
}

func (s *SatelliteUplink) BroadcastBlock(block types.Block) error {
    // Compress block (satellite bandwidth is limited)
    compressed := CompressBlock(block)
    
    // Encode as AX.25 packet
    packet := ax25.NewPacket(
        s.Callsign,
        "AEQUIT-1",  // Satellite identifier
        []byte(compressed),
    )
    
    // Transmit when satellite is overhead
    return s.Station.Send(packet)
}

func (s *SatelliteUplink) ReceiveBlocks() (<-chan types.Block, error) {
    blockChan := make(chan types.Block)
    
    // Listen for incoming packets
    s.Station.OnReceive(func(packet ax25.Packet) {
        if packet.Dest == "AEQUIT-1" {
            block := DecompressBlock(packet.Payload)
            blockChan <- block
        }
    })
    
    return blockChan, nil
}
