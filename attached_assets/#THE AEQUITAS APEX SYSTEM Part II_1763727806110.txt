class SwarmDrone:
    """Individual autonomous enforcement drone"""
    
    def __init__(self, drone_id: int, position: Tuple[float, float, float]):
        self.id = drone_id
        self.position = position  # (x, y, z)
        self.velocity = (0.0, 0.0, 0.0)
        self.target = None
        self.status = "IDLE"
        self.neighbors: Set[int] = set()
    
    def update_position(self, dt: float = 0.1):
        """Update drone position based on velocity"""
        self.position = tuple(
            p + v * dt for p, v in zip(self.position, self.velocity)
        )
    
    def set_target(self, target: Tuple[float, float, float]):
        """Set target location"""
        self.target = target
        self.status = "MOVING"
    
    def communicate(self, other_drones: List['SwarmDrone']):
        """Communicate with nearby drones (mesh network)"""
        self.neighbors.clear()
        
        for drone in other_drones:
            if drone.id != self.id:
                dist = self._distance_to(drone.position)
                if dist < 100.0:  # Communication range: 100m
                    self.neighbors.add(drone.id)
    
    def _distance_to(self, pos: Tuple[float, float, float]) -> float:
        """Calculate distance to position"""
        return sum((a - b) ** 2 for a, b in zip(self.position, pos)) ** 0.5

                drone._distance_to(self.mission['target']) < 10.0

class CommunicationSystem:
    """
    Multi-layer communication with fallbacks:
    1. Local Mesh Network (instant)
    2. Satellite (Starlink) (5-10s latency)
    3. LoRa Long-Range (20-30s latency)
    4. Cellular 5G (if available)
    5. Offline Queue (worst case)
    """
    
    def __init__(self):
        self.channels = {
            'mesh': {'status': 'ACTIVE', 'priority': 1},
            'satellite': {'status': 'STANDBY', 'priority': 2},
            'lora': {'status': 'STANDBY', 'priority': 3},
            'cellular': {'status': 'STANDBY', 'priority': 4},
            'offline': {'status': 'STANDBY', 'priority': 5}
        }
        
        self.message_queue = []
        
        logger.info("📡 Communication System initialized")
        logger.info(f"   Channels: {len(self.channels)}")
    
    def send_message(self, message: Dict, recipient: str) -> bool:
        """Send message with automatic fallback"""
        logger.info(f"📤 Sending message to {recipient}...")
        
        # Try channels in priority order
        for channel, config in sorted(self.channels.items(), key=lambda x: x[1]['priority']):
            if config['status'] in ['ACTIVE', 'STANDBY']:
                logger.info(f"   Trying {channel}...")
                
                if self._attempt_send(channel, message, recipient):
                    logger.info(f"✅ Message sent via {channel}")
                    return True
                else:
                    logger.warning(f"⚠️  {channel} failed, trying next...")
        
        # All channels failed - queue offline
        logger.warning("⚠️  All channels unavailable, queueing offline...")
        self.message_queue.append({'message': message, 'recipient': recipient, 'timestamp': time.time()})
        return False
    
    def _attempt_send(self, channel: str, message: Dict, recipient: str) -> bool:
        """Attempt to send via specific channel"""
        # Simulate different channel behaviors
        import random
        
        success_rates = {
            'mesh': 0.95,      # 95% reliable
            'satellite': 0.90,  # 90% reliable
            'lora': 0.85,      # 85% reliable
            'cellular': 0.80,   # 80% reliable
            'offline': 1.0     # Always succeeds (queues locally)
        }
        
        return random.random() < success_rates.get(channel, 0.5)
    
    def process_offline_queue(self):
        """Process queued messages when connectivity restored"""
        if not self.message_queue:
            return
        
        logger.info(f"📨 Processing {len(self.message_queue)} queued messages...")
        
        sent = 0
        failed = []
        
        for queued in self.message_queue:
            if self.send_message(queued['message'], queued['recipient']):
                sent += 1
            else:
                failed.append(queued)
        
        self.message_queue = failed
        logger.info(f"✅ Sent {sent} queued messages, {len(failed)} still pending")

        log_error "$1 not found. Installing..."
        return 1

    return 0
}

################################################################################
# SYSTEM REQUIREMENTS CHECK
################################################################################


check_system_requirements() {
    log_section "1. CHECKING SYSTEM REQUIREMENTS"
    
    # Check OS
    if [[ "$OSTYPE" == "linux-gnu"* ]]; then
        log_success "OS: Linux (supported)"
    elif [[ "$OSTYPE" == "darwin"* ]]; then
        log_warning "OS: macOS (limited support)"
    else
        log_error "Unsupported OS: $OSTYPE"
        exit 1
    fi
    
    # Check Python
    if check_command python3; then
        PYTHON_VERSION=$(python3 --version | awk '{print $2}')
        log_info "Python version: $PYTHON_VERSION"
    else
        log_error "Python 3.10+ required"
        exit 1
    fi
    
    # Check NVIDIA GPU
    if check_command nvidia-smi; then
        GPU_INFO=$(nvidia-smi --query-gpu=name,memory.total --format=csv,noheader | head -n 1)
        log_success "GPU detected: $GPU_INFO"
    else
        log_warning "No NVIDIA GPU detected (CPU-only mode)"
    fi
    
    # Check Docker
    if check_command docker; then
        DOCKER_VERSION=$(docker --version | awk '{print $3}')
        log_info "Docker version: $DOCKER_VERSION"
    else
        log_warning "Docker not installed (will install)"
    fi
    
    # Check Kubernetes
    if [[ "$MODE" == "cluster" ]] || [[ "$MODE" == "all" ]]; then
        if check_command kubectl; then
            KUBECTL_VERSION=$(kubectl version --client --short 2>/dev/null | awk '{print $3}')
            log_info "kubectl version: $KUBECTL_VERSION"
        else
            log_warning "kubectl not installed (will install)"
        fi
    fi
    
    # Check disk space (need 100GB+)
    AVAILABLE_SPACE=$(df -BG . | tail -1 | awk '{print $4}' | sed 's/G//')
    if [ "$AVAILABLE_SPACE" -lt 100 ]; then
        log_error "Insufficient disk space. Need 100GB+, have ${AVAILABLE_SPACE}GB"
        exit 1
    fi
    log_success "Disk space: ${AVAILABLE_SPACE}GB available"
    
    # Check RAM (need 32GB+)
    TOTAL_RAM=$(free -g | awk '/^Mem:/{print $2}')
    if [ "$TOTAL_RAM" -lt 32 ]; then
        log_warning "Low RAM: ${TOTAL_RAM}GB (recommended: 64GB+)"
    else
        log_success "RAM: ${TOTAL_RAM}GB"
    fi
}

install_cuda() {
    log_section "3. INSTALLING CUDA & NVIDIA DRIVERS"
    
    if ! check_command nvidia-smi; then
        log_info "Installing NVIDIA drivers and CUDA 12.1..."
        
        # Download CUDA installer
        wget https://developer.download.nvidia.com/compute/cuda/12.1.0/local_installers/cuda_12.1.0_530.30.02_linux.run
        
        # Install CUDA
        sudo sh cuda_12.1.0_530.30.02_linux.run --silent --toolkit
        
        # Add to PATH
        echo 'export PATH=/usr/local/cuda-12.1/bin:$PATH' >> ~/.bashrc
        echo 'export LD_LIBRARY_PATH=/usr/local/cuda-12.1/lib64:$LD_LIBRARY_PATH' >> ~/.bashrc
        source ~/.bashrc
        
        log_success "CUDA 12.1 installed"
    else
        log_success "CUDA already installed"
    fi
}

install_ros2() {
    log_section "5. INSTALLING ROS2 HUMBLE (FOR SWARM ROBOTICS)"
    
    if check_command ros2; then
        log_success "ROS2 already installed"
        return
    fi
    
    log_info "Installing ROS2 Humble..."
    
    # Add ROS2 apt repository
    sudo apt install software-properties-common
    sudo add-apt-repository universe
    
    sudo apt update && sudo apt install curl -y
    sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key -o /usr/share/keyrings/ros-archive-keyring.gpg
    
    echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" | sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null
    
    # Install ROS2 Humble
    sudo apt update
    sudo apt install -y ros-humble-desktop
    sudo apt install -y ros-dev-tools
    
    # Source ROS2
    echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc
    source /opt/ros/humble/setup.bash
    
    log_success "ROS2 Humble installed"
}

setup_blockchain() {
    log_section "7. SETTING UP LOCAL BLOCKCHAIN NODE"
    
    mkdir -p "$BLOCKCHAIN_DIR"
    cd "$BLOCKCHAIN_DIR"
    
    # Install Go (for Tendermint)
    if ! check_command go; then

        wget https://go.dev/dl/go1.21.0.linux-amd64.tar.gz
        sudo tar -C /usr/local -xzf go1.21.0.linux-amd64.tar.gz
        echo 'export PATH=$PATH:/usr/local/go/bin' >> ~/.bashrc
        source ~/.bashrc
    fi
    
    # Install Tendermint
    if ! check_command tendermint; then
        log_info "Installing Tendermint..."
        git clone https://github.com/tendermint/tendermint.git
        cd tendermint
        make install
        cd ..
    fi
    
    # Initialize Tendermint
    if [ ! -d "$BLOCKCHAIN_DIR/.tendermint" ]; then
        log_info "Initializing Tendermint..."
        tendermint init
    fi
    
    log_success "Blockchain node configured"
}

build_docker_images() {
    log_section "9. BUILDING DOCKER IMAGES"
    
    mkdir -p "$APEX_DIR/docker"
    
    # Main APEX Dockerfile
    cat > "$APEX_DIR/docker/Dockerfile.apex" << 'EOF'
FROM nvidia/cuda:12.1.0-devel-ubuntu22.04


# Install system dependencies
RUN apt-get update && apt-get install -y \
    python3.10 \
    python3-pip \
    git \
    git-lfs \
    wget \
    curl \
    build-essential \
    cmake \
    && rm -rf /var/lib/apt/lists/*


# Install Python packages
RUN pip3 install --no-cache-dir \
    torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121 \
    transformers==4.36.0 \
    accelerate==0.25.0 \
    bitsandbytes==0.41.3 \
    liboqs-python \
    web3 \
    requests \
    vllm


# Copy application code
COPY apex_system.py /app/apex_system.py
COPY requirements.txt /app/requirements.txt


WORKDIR /app


# Install additional requirements
RUN pip3 install --no-cache-dir -r requirements.txt


# Set entrypoint
ENTRYPOINT ["python3", "apex_system.py"]
EOF
    
    # Swarm Dockerfile
    cat > "$APEX_DIR/docker/Dockerfile.swarm" << 'EOF'
FROM ros:humble


# Install Python and dependencies
RUN apt-get update && apt-get install -y \
    python3-pip \
    && rm -rf /var/lib/apt/lists/*


RUN pip3 install --no-cache-dir \
    numpy \
    scipy


# Copy swarm controller code
COPY swarm_controller.py /app/swarm_controller.py


WORKDIR /app


CMD ["python3", "swarm_controller.py"]
EOF
    
    # Build images
    if check_command docker; then
        log_info "Building APEX AI image..."
        docker build -t aequitas/apex:latest -f "$APEX_DIR/docker/Dockerfile.apex" "$APEX_DIR"
        
        log_info "Building Swarm Controller image..."
        docker build -t aequitas/apex-swarm:latest -f "$APEX_DIR/docker/Dockerfile.swarm" "$APEX_DIR"
        
        log_success "Docker images built"
    else
        log_warning "Docker not available, skipping image build"
    fi

start_local_services() {
    log_section "11. STARTING LOCAL SERVICES"
    
    mkdir -p "$LOGS_DIR"
    
    # Start blockchain node
    log_info "Starting blockchain node..."
    cd "$BLOCKCHAIN_DIR"
    nohup tendermint node > "$LOGS_DIR/blockchain.log" 2>&1 &
    BLOCKCHAIN_PID=$!
    echo $BLOCKCHAIN_PID > "$APEX_DIR/blockchain.pid"
    log_success "Blockchain node started (PID: $BLOCKCHAIN_PID)"
    
    # Start APEX system
    log_info "Starting APEX system..."
    cd "$APEX_DIR"
    source venv/bin/activate
    nohup python3 apex_system.py > "$LOGS_DIR/apex.log" 2>&1 &
    APEX_PID=$!
    echo $APEX_PID > "$APEX_DIR/apex.pid"
    log_success "APEX system started (PID: $APEX_PID)"
    
    echo ""
    log_info "Services running:"
    log_info "  Blockchain: PID $BLOCKCHAIN_PID (logs: $LOGS_DIR/blockchain.log)"
    log_info "  APEX:       PID $APEX_PID (logs: $LOGS_DIR/apex.log)"
    echo ""
    log_info "To stop services: kill $BLOCKCHAIN_PID $APEX_PID"
}

class LegalDocumentParser:
    """
    Parses legal documents into structured format
    
    Supports:
    - PDF (with OCR for scanned docs)
    - HTML (web-scraped content)
    - XML (structured legal data)
    - Plain text
    """
    
    def __init__(self):
        # Load spaCy model for NER (Named Entity Recognition)
        try:
            self.nlp = spacy.load('en_core_web_lg')
        except:
            logger.warning("⚠️  spaCy model not found. Run: python -m spacy download en_core_web_lg")
            self.nlp = None
        
        logger.info("📄 Legal Document Parser initialized")
    
    def parse_pdf(self, pdf_path: Path) -> str:
        """Parse PDF document"""
        if not LEGAL_PROCESSING_AVAILABLE:
            logger.error("pdfplumber not installed")
            return ""
        
        text = ""
        
        try:
            with pdfplumber.open(pdf_path) as pdf:
                for page in pdf.pages:
                    text += page.extract_text() + "\n\n"
            
            logger.info(f"✅ Parsed PDF: {pdf_path.name} ({len(text)} chars)")
            
        except Exception as e:
            logger.error(f"❌ Failed to parse PDF: {e}")
        
        return text
    
    def parse_html(self, html_path: Path) -> str:
        """Parse HTML document"""
        try:
            html_content = html_path.read_text()
            soup = BeautifulSoup(html_content, 'html.parser')
            
            # Remove scripts and styles
            for script in soup(["script", "style"]):
                script.decompose()
            
            text = soup.get_text()
            
            # Clean up whitespace
            lines = (line.strip() for line in text.splitlines())
            chunks = (phrase.strip() for line in lines for phrase in line.split("  "))
            text = '\n'.join(chunk for chunk in chunks if chunk)
            
            logger.info(f"✅ Parsed HTML: {html_path.name} ({len(text)} chars)")
            
            return text
            
        except Exception as e:
            logger.error(f"❌ Failed to parse HTML: {e}")
            return ""
    
    def extract_citations(self, text: str) -> List[str]:
        """Extract legal citations from text"""
        citations = []
        
        # Common citation patterns
        patterns = [
            r'\d+\s+U\.S\.\s+\d+',  # US Supreme Court
            r'\d+\s+F\.\d+d\s+\d+',  # Federal cases
            r'\d+\s+U\.S\.C\.\s+§\s+\d+',  # US Code
            r'\d+\s+Stat\.\s+\d+',  # Statutes at Large
        ]
        
        for pattern in patterns:
            matches = re.findall(pattern, text)
            citations.extend(matches)
        
        return list(set(citations))  # Remove duplicates
    
    def extract_legal_entities(self, text: str) -> Dict[str, List[str]]:
        """Extract legal entities (parties, judges, courts)"""
        if not self.nlp:
            return {}
        
        doc = self.nlp(text[:1000000])  # Limit to 1M chars for performance
        
        entities = {
            'PERSON': [],
            'ORG': [],
            'GPE': [],  # Geopolitical entities
            'LAW': [],
            'DATE': []
        }
        
        for ent in doc.ents:
            if ent.label_ in entities:
                entities[ent.label_].append(ent.text)
        
        # Remove duplicates
        for key in entities:
            entities[key] = list(set(entities[key]))
        
        return entities

class LegalKnowledgeDatabase:
    """
    Vector database for legal knowledge
    
    Uses ChromaDB for semantic search across legal corpus
    Enables:
    - Find relevant laws by natural language query
    - Cross-reference across jurisdictions
    - Similarity search for precedents
    """
    
    def __init__(self, db_path: Path):
        self.db_path = db_path
        self.db_path.mkdir(parents=True, exist_ok=True)
        
        if not LEGAL_PROCESSING_AVAILABLE:
            logger.error("ChromaDB not available")
            self.client = None
            return
        
        # Initialize ChromaDB
        self.client = chromadb.PersistentClient(
            path=str(db_path),
            settings=Settings(anonymized_telemetry=False)
        )
        
        # Create collections
        self.collections = {
            'statutes': self.client.get_or_create_collection('statutes'),
            'cases': self.client.get_or_create_collection('cases'),
            'treaties': self.client.get_or_create_collection('treaties'),
            'constitutions': self.client.get_or_create_collection('constitutions'),
        }
        
        logger.info("🗄️  Legal Knowledge Database initialized")
        logger.info(f"   Database path: {db_path}")
        logger.info(f"   Collections: {len(self.collections)}")
    
    def ingest_document(self, collection_name: str, doc_id: str, text: str, metadata: Dict):
        """Ingest document into vector database"""
        if not self.client:
            return
        
        collection = self.collections.get(collection_name)
        if not collection:
            logger.error(f"Collection {collection_name} not found")
            return
        
        # Chunk document (ChromaDB has size limits)
        chunks = self._chunk_text(text, max_chars=5000)
        
        for i, chunk in enumerate(chunks):
            chunk_id = f"{doc_id}_chunk_{i}"
            
            collection.add(
                ids=[chunk_id],
                documents=[chunk],
                metadatas=[{**metadata, 'chunk': i, 'total_chunks': len(chunks)}]
            )
        
        logger.info(f"✅ Ingested: {doc_id} ({len(chunks)} chunks)")
    
    def _chunk_text(self, text: str, max_chars: int = 5000) -> List[str]:
        """Chunk text into smaller pieces"""
        chunks = []
        current_chunk = ""
        
        for paragraph in text.split('\n\n'):
            if len(current_chunk) + len(paragraph) > max_chars:
                if current_chunk:
                    chunks.append(current_chunk)
                current_chunk = paragraph
            else:
                current_chunk += "\n\n" + paragraph
        
        if current_chunk:
            chunks.append(current_chunk)
        
        return chunks
    
    def search(self, collection_name: str, query: str, n_results: int = 10) -> List[Dict]:
        """Search legal database"""
        if not self.client:
            return []
        
        collection = self.collections.get(collection_name)
        if not collection:
            return []
        
        results = collection.query(
            query_texts=[query],
            n_results=n_results
        )
        
        return results

    def find_loopholes(self, jurisdiction: str, topic: str) -> List[Loophole]:
        """
        Find legal loopholes that prevent remedy
        
        Examples of loopholes:
        - Statute of limitations that bar claims
        - Sovereign immunity doctrines
        - Standing requirements
        - Procedural bars (laches, estoppel)
        - Forced arbitration clauses
        """
        logger.info(f"🔍 Searching for loopholes: {jurisdiction} - {topic}")
        
        loopholes = []
        
        # Search relevant laws
        search_query = f"{topic} {jurisdiction} immunity statute limitations standing procedural bar"
        
        # Search across all collections
        for collection_name in ['statutes', 'cases']:
            results = self.database.search(collection_name, search_query, n_results=20)
            
            if not results or 'documents' not in results:
                continue
            
            for i, doc in enumerate(results['documents'][0]):
                metadata = results['metadatas'][0][i] if 'metadatas' in results else {}
                
                # Analyze for remedy-blocking provisions
                loophole_indicators = [
                    'statute of limitations',
                    'sovereign immunity',
                    'qualified immunity',
                    'lack of standing',
                    'laches',
                    'estoppel',
                    'mandatory arbitration',
                    'procedural bar',
                    'time-barred',
                    'jurisdictional bar'
                ]
                
                for indicator in loophole_indicators:
                    if indicator.lower() in doc.lower():
                        loophole = Loophole(
                            law=metadata.get('type', 'unknown'),
                            citation=metadata.get('citation', 'N/A'),
                            jurisdiction=metadata.get('jurisdiction', jurisdiction),
                            loophole_type=indicator.replace(' ', '_'),
                            text=doc[:500],  # First 500 chars
                            remedy_blocked=f"Blocks claims based on {indicator}",
                            closure_strategy=self._generate_closure_strategy(indicator, jurisdiction)
                        )
                        
                        loopholes.append(loophole)
                        break  # One loophole per document
        
        self.loopholes_found.extend(loopholes)
        
        logger.info(f"✅ Found {len(loopholes)} potential loopholes")
        
        return loopholes

        """Generate strategy to close a legal loophole"""
        strategies = {
            'statute of limitations': """

                2. Apply equitable tolling (discovery rule)
                3. Invoke fraudulent concealment exception
                4. Cite international law (no statute of limitations for genocide/jus cogens violations)
                5. Challenge constitutionality under due process
            """,

                4. Jus cogens violations override immunity
                5. Treaty obligations waive immunity
                6. Constitutional violations pierce immunity
            """,

    def generate_legal_argument(self, claim: str, jurisdiction: str, context: Optional[Dict] = None) -> Dict:

    def _assess_argument_strength(self, constitutional: List, precedents: List, statutes: List) -> Dict:
        """Assess strength of legal argument"""
        
        score = 0
        max_score = 100
        
        # Constitutional support (40 points)
        if constitutional:
            score += min(40, len(constitutional) * 10)
        
        # Case precedents (30 points)
        if precedents:
            score += min(30, len(precedents) * 3)
        
        # Statutory support (20 points)
        if statutes:
            score += min(20, len(statutes) * 5)
        
        # Multiple jurisdiction support (10 points)
        jurisdictions = set()
        for item in constitutional + precedents + statutes:
            if isinstance(item, dict) and 'jurisdiction' in item:
                jurisdictions.add(item['jurisdiction'])
        
        if len(jurisdictions) > 1:
            score += min(10, len(jurisdictions) * 2)
        
        # Assessment
        if score >= 80:
            assessment = "VERY STRONG - High likelihood of success"
        elif score >= 60:
            assessment = "STRONG - Likely to succeed"
        elif score >= 40:
            assessment = "MODERATE - Reasonable chance of success"
        else:
            assessment = "WEAK - May face challenges"
        
        return {
            'score': score,
            'max_score': max_score,
            'percentage': (score / max_score) * 100,
            'assessment': assessment,
            'constitutional_support': len(constitutional),
            'case_precedents': len(precedents),
            'statutory_support': len(statutes),
            'jurisdictional_breadth': len(jurisdictions)
        }

        for jurisdiction in jurisdictions:

            # Generate argument for this jurisdiction
            argument = self.generate_legal_argument(claim, jurisdiction)
            
            results[jurisdiction] = {

                'statutory_support': len(argument['statutory_support']),

        # Find universal principles
        universal = self._identify_universal_principles(results)
        
        return {
            'jurisdictions_analyzed': jurisdictions,
            'results_by_jurisdiction': results,
            'conflicts': conflicts,
            'universal_principles': universal,
            'strongest_jurisdiction': max(results.items(), key=lambda x: x[1]['strength']['score'])[0],
            'weakest_jurisdiction': min(results.items(), key=lambda x: x[1]['strength']['score'])[0]
        }
    
    def _identify_jurisdictional_conflicts(self, results: Dict) -> List[Dict]:
        """Identify conflicts between jurisdictions"""
        conflicts = []
        
        
        # Compare each jurisdiction pair
        jurisdictions = list(results.keys())
        
        for i in range(len(jurisdictions)):
            for j in range(i + 1, len(jurisdictions)):
                jur1 = jurisdictions[i]
                jur2 = jurisdictions[j]
                
                strength_diff = abs(
                    results[jur1]['strength']['score'] - 
                    results[jur2]['strength']['score']
                )
                
                # Significant difference indicates potential conflict
                if strength_diff > 30:
                    conflicts.append({
                        'jurisdiction_1': jur1,
                        'jurisdiction_2': jur2,
                        'strength_difference': strength_diff,
                        'analysis': f"{jur1} provides {strength_diff} points stronger support than {jur2}"
                    })
        
        return conflicts
    
    def _identify_universal_principles(self, results: Dict) -> List[str]:
        """Identify universal legal principles across jurisdictions"""
        universal_principles = []
        
        # Check for international treaty support across all jurisdictions

        if all(result.get('case_precedents', 0) > 2 for result in results.values()):
            universal_principles.append("Strong precedential support across all jurisdictions")
        
        if all(result.get('strength', {}).get('score', 0) > 60 for result in results.values()):
            universal_principles.append("Claim has strong legal basis in all jurisdictions analyzed")
        
        return universal_principles

    def validate_block(self, block: Dict) -> Tuple[bool, str]:
        """
        Validate block according to node role
        
        Each node applies its specialized validation logic
        Returns: (is_valid, reason)
        """
        self.votes_cast += 1
        
        if self.role == NodeRole.GENESIS:
            return self._validate_genesis(block)
        elif self.role == NodeRole.APEX:
            return self._validate_apex(block)
        elif self.role == NodeRole.SECURE:
            return self._validate_secure(block)
        elif self.role == NodeRole.SENTINEL:
            return self._validate_sentinel(block)
        elif self.role == NodeRole.ARBITER:
            return self._validate_arbiter(block)
        elif self.role == NodeRole.ORACLE:
            return self._validate_oracle(block)
        elif self.role == NodeRole.ARCHIVE:
            return self._validate_archive(block)
    
    def _validate_genesis(self, block: Dict) -> Tuple[bool, str]:
        """Genesis node validation"""
        # Check constitutional compliance
        if not self._check_constitutional(block):
            self.disagreements += 1
            return False, "Constitutional violation detected"
        
        # Check REPAR distribution rules
        if 'repar_transfer' in block:
            if not self._validate_repar_transfer(block['repar_transfer']):
                self.disagreements += 1
                return False, "Invalid REPAR transfer"
        
        self.blocks_validated += 1
        return True, "Genesis validation passed"
    
    def _validate_apex(self, block: Dict) -> Tuple[bool, str]:
        """APEX node validation"""
        # Check AI-determined validity
        if not self._check_ai_consensus(block):
            self.disagreements += 1
            return False, "AI consensus failed"
        
        # Check security posture
        if not self._check_security(block):
            self.disagreements += 1
            return False, "Security threat detected"
        
        self.blocks_validated += 1
        return True, "APEX validation passed"
    
    def _validate_secure(self, block: Dict) -> Tuple[bool, str]:
        """SECURE node validation"""
        # Check legal compliance
        if not self._check_legal_compliance(block):
            self.disagreements += 1
            return False, "Legal compliance failed"
        
        # Check for loopholes being exploited
        loopholes = self.legal_system.find_loopholes(
            block.get('jurisdiction', 'US'),
            block.get('action_type', 'unknown')
        )
        
        if loopholes:
            self.disagreements += 1
            return False, f"Legal loophole detected: {loopholes[0].loophole_type}"
        
        self.blocks_validated += 1
        return True, "SECURE validation passed"
    
    def _validate_sentinel(self, block: Dict) -> Tuple[bool, str]:
        """SENTINEL node validation (NEW)"""
        # Check for attack patterns
        if self._detect_attack_pattern(block):
            self.disagreements += 1
            self.threat_database['security_incidents'].append({
                'block_id': block.get('id'),
                'timestamp': time.time(),
                'threat_type': 'attack_pattern_detected'
            })
            return False, "Attack pattern detected"
        
        # Check transaction velocity
        if block.get('transaction_count', 0) > self.thresholds['rapid_transactions']:
            self.disagreements += 1
            return False, f"Abnormal transaction velocity: {block.get('transaction_count')}/sec"
        
        # Check transfer amounts
        if 'repar_transfer' in block:
            amount = block['repar_transfer'].get('amount', 0)
            if amount > self.thresholds['large_transfers']:
                logger.warning(f"⚠️  Large transfer detected: {amount} REPAR")
                # Not necessarily invalid, but flagged for review
        
        self.blocks_validated += 1
        return True, "SENTINEL validation passed"
    
    def _validate_arbiter(self, block: Dict) -> Tuple[bool, str]:
        """ARBITER node validation (NEW)"""
        # Check for conflicts with other nodes
        if 'validation_conflict' in block:
            # Arbiter mediates disagreements
            conflict = block['validation_conflict']
            resolution = self._resolve_conflict(conflict)
            
            if not resolution['resolved']:
                self.disagreements += 1
                return False, f"Unresolvable conflict: {resolution['reason']}"
        
        # Check fairness of action
        if not self._check_fairness(block):
            self.disagreements += 1
            return False, "Action violates fairness principles"
        
        # Check consistency with precedent
        if not self._check_consistency(block):
            self.disagreements += 1
            return False, "Inconsistent with established precedent"
        
        self.blocks_validated += 1
        return True, "ARBITER validation passed"
    
    def _validate_oracle(self, block: Dict) -> Tuple[bool, str]:
        """ORACLE node validation (NEW)"""
        # Verify external data references
        if 'external_data' in block:
            for data_ref in block['external_data']:
                if not self._verify_external_data(data_ref):
                    self.disagreements += 1
                    return False, f"External data verification failed: {data_ref.get('source')}"
        
        # Verify price feeds (if applicable)
        if 'price_data' in block:
            if not self._verify_price_feed(block['price_data']):
                self.disagreements += 1
                return False, "Price feed verification failed"
        
        # Verify document authenticity
        if 'documents' in block:
            for doc in block['documents']:
                if not self._verify_document_authenticity(doc):
                    self.disagreements += 1
                    return False, f"Document authenticity failed: {doc.get('hash')}"
        
        self.blocks_validated += 1
        return True, "ORACLE validation passed"
    
    def _validate_archive(self, block: Dict) -> Tuple[bool, str]:
        """ARCHIVE node validation (NEW)"""
        # Check block is properly formatted for archiving
        if not self._check_archival_format(block):
            self.disagreements += 1
            return False, "Block format unsuitable for archival"
        
        # Verify cryptographic integrity
        if not self._verify_cryptographic_integrity(block):
            self.disagreements += 1
            return False, "Cryptographic integrity check failed"
        
        # Check temporal consistency
        if not self._check_temporal_consistency(block):
            self.disagreements += 1
            return False, "Temporal inconsistency detected"
        
        # Archive the block
        self._archive_block(block)
        
        self.blocks_validated += 1
        return True, "ARCHIVE validation passed"

        """Verify cryptographic integrity"""
        if 'hash' not in block:
            return False
        
        # Recompute hash
        block_copy = {k: v for k, v in block.items() if k != 'hash'}
        computed_hash = hashlib.sha256(
            json.dumps(block_copy, sort_keys=True).encode()
        ).hexdigest()
        
        return computed_hash == block.get('hash')

        """Check temporal consistency"""
        if 'timestamp' not in block:
            return False
        
        block_time = block['timestamp']
        current_time = time.time()
        
        # Block shouldn't be from the future
        if block_time > current_time + 300:  # 5 min tolerance
            return False
        
        return True
    
    def _archive_block(self, block: Dict):
        """Archive block to permanent storage"""
        block_id = block.get('id', 'unknown')
        timestamp = block.get('timestamp', time.time())
        
        # Determine storage tier based on age
        age_seconds = time.time() - timestamp
        
        if age_seconds < 86400:  # < 1 day
            tier = 'hot'
        elif age_seconds < 15552000:  # < 6 months
            tier = 'warm'
        elif age_seconds < 31536000:  # < 1 year
            tier = 'cold'
        else:
            tier = 'glacier'
        
        # Save to archive
        archive_file = self.archive_path / tier / f'block_{block_id}.json'
        archive_file.parent.mkdir(parents=True, exist_ok=True)
        
        archive_file.write_text(json.dumps(block, indent=2))
        
        # Update stats
        self.archive_stats['total_records'] += 1
        self.archive_stats['total_size_bytes'] += len(json.dumps(block))
        
        if self.archive_stats['oldest_record'] is None:
            self.archive_stats['oldest_record'] = timestamp
        
        self.archive_stats['newest_record'] = timestamp
        
        logger.debug(f"   Archived block {block_id} to {tier} storage")

            node = self.nodes[role]
            
            logger.info(f"\n[{role.name}] Validating...")
            
            try:
                is_valid, reason = node.validate_block(block)
                validations[role] = (is_valid, reason)
                
                status = "✅ APPROVE" if is_valid else "❌ REJECT"
                logger.info(f"  {status}: {reason}")
                
            except Exception as e:
                logger.error(f"  ❌ VALIDATION ERROR: {e}")
                validations[role] = (False, f"Error: {str(e)}")
        
        # Count approvals
        approvals = sum(1 for valid, _ in validations.values() if valid)
        rejections = len(validations) - approvals
        
        # Check consensus (5 of 7)
        consensus_reached = approvals >= self.consensus_threshold
        
        logger.info("\n" + "=" * 80)
        logger.info("CONSENSUS RESULT")
        logger.info("=" * 80)
        logger.info(f"Approvals: {approvals}/{len(validations)}")
        logger.info(f"Rejections: {rejections}/{len(validations)}")
        logger.info(f"Threshold: {self.consensus_threshold}/{self.total_nodes}")
        
        if consensus_reached:
            logger.info(f"\n✅ CONSENSUS REACHED ({approvals}/{len(validations)} approved)")
            self.blocks_accepted += 1
            
            result = {
                'consensus_reached': True,
                'approvals': approvals,
                'rejections': rejections,
                'total_votes': len(validations),
                'threshold': self.consensus_threshold,
                'validations': {role.name: {'valid': v, 'reason': r} for role, (v, r) in validations.items()},
                'block_accepted': True
            }
            
        else:
            logger.error(f"\n❌ CONSENSUS FAILED ({approvals}/{len(validations)} approved, need {self.consensus_threshold})")
            self.blocks_rejected += 1
            self.consensus_failures += 1
            
            result = {
                'consensus_reached': False,
                'approvals': approvals,
                'rejections': rejections,
                'total_votes': len(validations),
                'threshold': self.consensus_threshold,
                'validations': {role.name: {'valid': v, 'reason': r} for role, (v, r) in validations.items()},
                'block_accepted': False,
                'rejection_reasons': [r for v, r in validations.values() if not v]
            }
        
        logger.info("=" * 80)
        
        return result
    
    def simulate_node_failure(self, role: NodeRole):

        self.node_health[role] = False
        logger.warning(f"⚠️  NODE OFFLINE: {role.name}")

        else:

        logger.info(f"✅ NODE RESTORED: {role.name}")
        
        online_count = sum(1 for healthy in self.node_health.values() if healthy)
        logger.info(f"   Nodes online: {online_count}/{self.total_nodes}")
    
    def get_network_status(self) -> Dict:
        """Get comprehensive network status"""
        online_nodes = sum(1 for healthy in self.node_health.values() if healthy)
        
        return {

            },
            'network_status': self._get_network_status_text(online_nodes),

    def _get_network_status_text(self, online_nodes: int) -> str:
